# Lexicon of the .intent format: the approved words and the roles each type presupposes.
# One entry per line: =class id | field: value | ... ; lists use ", ", maps use "key=value",
# a boolean field is the bare word. Extensions in a project's .intent files use the same lines.
# A verb with `target: literal|entity|time` requires that kind of object after the verb: the check
# errors on a literal where the verb expects an entity, or the other way round.

=version 0.3

# ---------------------------------------------------------------- grammatical core
=core article | en: the
=core conditional | if: se | when: quando | while: enquanto
=core negation | en: does not, never
=core universal | en: always
# declared exception: the relation that goes up a layer on purpose
=core exception | en: exceptionally
# quantifier: Every <type> in the <type> "<name>" <predicate>.
=core quantifier | en: every
=core coordination | en: and
# prepositions and copula
=core preposition | en: from, to, for, into, through, on, with, in, onto
=core copula | en: is
# negated copula: inequality (X is not the value Y) and, before a verb with "is", the negation
# of that verb (is not in, is not keyed by), generated from the verb itself
=core negated_copula | en: is not
